1//! Error types for Nintendo EAD/SRD Super Mario Advance parsing and synthesis.
2
3use core::fmt;
4
5/// Errors returned when parsing Nintendo EAD/SRD sample descriptors or
6/// synthesis parameters.
7#[derive(Debug, Clone, PartialEq, Eq, Hash)]
8#[non_exhaustive]
9pub enum Error {
10    /// Input buffer ended before a complete header or payload could be read.
11    UnexpectedEof,
12    /// Descriptor or container header failed structural validation.
13    InvalidHeader(&'static str),
14    /// Loop range was invalid for the declared sample length.
15    InvalidLoopRange {
16        /// Loop start offset in samples.
17        loop_start: u32,
18        /// Loop end offset in samples.
19        loop_end: u32,
20        /// Total sample length in samples.
21        sample_len: u32,
22    },
23    /// Sample index was out of range.
24    InvalidSample(u32),
25    /// Invalid synthesis parameter.
26    InvalidParameter(&'static str),
27}
28
29impl fmt::Display for Error {
30    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
31        match self {
32            Self::UnexpectedEof => {
33                write!(f, "Unexpected end of data while parsing EAD/SMA stream")
34            }
35            Self::InvalidHeader(msg) => write!(f, "Invalid EAD/SMA header: {msg}"),
36            Self::InvalidLoopRange {
37                loop_start,
38                loop_end,
39                sample_len,
40            } => write!(
41                f,
42                "Invalid EAD/SMA loop range {loop_start}..{loop_end} for {sample_len}-byte sample"
43            ),
44            Self::InvalidSample(id) => write!(f, "Invalid EAD/SMA sample index {id}"),
45            Self::InvalidParameter(msg) => write!(f, "Invalid EAD/SMA parameter: {msg}"),
46        }
47    }
48}
49
50#[cfg(feature = "std")]
51impl std::error::Error for Error {}
52
53/// Convenience result alias for `ead-sma`.
54pub type Result<T> = core::result::Result<T, Error>;
