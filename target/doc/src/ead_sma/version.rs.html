1//! Nintendo EAD/SRD Super Mario Advance engine revision detection.
2//!
3//! A full-corpus audit of all 3,377 GBA ROMs identifies **35 ROM images**
4//! powered by the EAD/SRD custom DirectSound mixer across **2 distinct
5//! ARM7TDMI IWRAM kernel revisions**:
6//!
7//! 1. [`EadSmaVersion::Sma23Shift8`] (**12 ROMs** — *Super Mario Advance 2*
8//!    USA/Europe/Japan/China, *Super Mario Advance 3* USA/Europe/Japan in
9//!    retail and Virtual Console forms, the China-market *Yoshi's Island*
10//!    release, and a China-market *Mario & Luigi* prototype whose embedded
11//!    *Mario Bros.* classic mode uses this driver):
12//!    - Voice mixer accumulates `(sample * volume) >> 8` per output frame
13//!      (ARM `mul` + `add ..., asr #8`, truncated to `i16` on store).
14//!    - DMA converter stores the low byte of each `i16` accumulator,
15//!      saturating to `[-128, 127]` only when the value lies outside that
16//!      range (`(v & 0xFF80) != 0 && != 0xFF80`).
17//!    - Buffer combine attenuates with round-half bias `1 << (shift - 1)`
18//!      for **all** source values, negative or not.
19//! 2. [`EadSmaVersion::Sma4Mla`] (**23 ROMs** — all *Super Mario Advance 4*
20//!    regional/revision/Virtual Console/Switch Online images, the
21//!    *Mario Bros.* classic mode inside *Super Mario Advance 3*, and
22//!    *The Legend of Zelda: A Link to the Past & Four Swords* USA/Europe/Japan):
23//!    - Voice mixer accumulates `sample * volume` with **no shift**
24//!      (ARM `mla`), so its voice volumes are pre-scaled by 1/256 relative
25//!      to the older revision.
26//!    - DMA converter first computes `(v + (v < 0 ? 127 : 0)) >> 7`
27//!      (round-toward-zero division by 128), then applies the same
28//!      `[-128, 127]` saturation test to the shifted value.
29//!    - Buffer combine attenuates with bias `(1 << shift) - 1` applied
30//!      **only to negative** source values (ceil-style rounding).
31//!
32//! Both revisions share the same voice-render wrapper semantics (Q8 phase,
33//! `176`-sample DMA frames, planar left/right `i16` mix buffers), the same
34//! 16-byte sample descriptor layout, and identical ARM unsigned/signed
35//! division helpers. *Super Mario Advance 3* cartridges contain **both**
36//! revisions (the main game uses `Sma23Shift8`; the *Mario Bros.* classic
37//! mode uses `Sma4Mla`); [`EadSmaVersion::detect_from_bytes`] reports the
38//! main-game revision in that case.
39
40/// Distinct Nintendo EAD/SRD ARM7TDMI mixer revisions discovered across the GBA corpus.
41#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Default)]
42pub enum EadSmaVersion {
43    /// SMA2/SMA3 revision (12 ROMs): mixer accumulates `(sample * volume) >> 8`;
44    /// DMA converter saturates the raw `i16` accumulator; combine uses the
45    /// unconditional round-half bias `1 << (shift - 1)`.
46    #[default]
47    Sma23Shift8,
48    /// SMA4 revision (23 ROMs): mixer accumulates `sample * volume` via `mla`
49    /// with no shift; DMA converter pre-scales by `>> 7` with negative bias
50    /// `+127` before saturating; combine biases only negative values by
51    /// `(1 << shift) - 1`.
52    Sma4Mla,
53}
54
55impl EadSmaVersion {
56    /// Returns `true` when the voice mixer shifts each product right by 8
57    /// before accumulating (`Sma23Shift8`), or `false` when it accumulates
58    /// the full product (`Sma4Mla`).
59    #[inline]
60    #[must_use]
61    pub const fn mixer_shifts_product(self) -> bool {
62        match self {
63            Self::Sma23Shift8 => true,
64            Self::Sma4Mla => false,
65        }
66    }
67
68    /// Number of GBA ROM images in the audited corpus using this revision.
69    #[inline]
70    #[must_use]
71    pub const fn corpus_rom_count(self) -> usize {
72        match self {
73            Self::Sma23Shift8 => 12,
74            Self::Sma4Mla => 23,
75        }
76    }
77
78    /// Detect the exact Nintendo EAD/SRD mixer revision from a GBA ROM image.
79    ///
80    /// Matches the three ARM7TDMI IWRAM kernel signatures (DMA converter,
81    /// voice mixer, buffer combine) at their revision-specific relative
82    /// offsets. The kernels are copied verbatim from ROM into IWRAM by the
83    /// retail driver, so the ROM signatures are byte-identical to the
84    /// executing code. Returns `None` for ROMs that do not contain this
85    /// engine (including all M4A titles such as the first
86    /// *Super Mario Advance*, *Mario Kart: Super Circuit*, and *F-Zero*).
87    #[must_use]
88    pub fn detect_from_bytes(rom: &[u8]) -> Option<Self> {
89        // Sma23Shift8 kernel block: [converter][mixer @ +0xC0][combine @ +0x1E0]
90        const A_CONV: [u8; 24] = [
91            0x30, 0x00, 0x2D, 0xE9, 0x7F, 0x40, 0xA0, 0xE3, 0xFF, 0x5C, 0xA0, 0xE3, 0x80, 0x50,
92            0x85, 0xE3, 0xF2, 0x30, 0xD0, 0xE0, 0x05, 0xC0, 0x13, 0xE0,
93        ];
94        const A_MIX: [u8; 24] = [
95            0x0D, 0xC0, 0xA0, 0xE1, 0xF0, 0x03, 0x2D, 0xE9, 0xF0, 0x00, 0x9C, 0xE8, 0x01, 0xC0,
96            0x43, 0xE0, 0x07, 0xC0, 0x0C, 0xE2, 0x00, 0x00, 0x5C, 0xE3,
97        ];
98        const A_COMB: [u8; 24] = [
99            0x30, 0x00, 0x2D, 0xE9, 0x01, 0x40, 0xA0, 0xE3, 0x01, 0xC0, 0x43, 0xE2, 0x14, 0x4C,
100            0xA0, 0xE1, 0xF0, 0x50, 0xD0, 0xE1, 0xF0, 0xC0, 0xD1, 0xE1,
101        ];
102        // Sma4Mla kernel block: [converter][mixer @ +0x120][combine @ +0x218]
103        const B_CONV: [u8; 24] = [
104            0x30, 0x00, 0x2D, 0xE9, 0x7F, 0x40, 0xA0, 0xE3, 0xFF, 0x5C, 0xA0, 0xE3, 0x80, 0x50,
105            0x85, 0xE3, 0xF2, 0x30, 0xD0, 0xE0, 0x00, 0x00, 0x53, 0xE3,
106        ];
107        const B_MIX: [u8; 24] = [
108            0x0D, 0xC0, 0xA0, 0xE1, 0xF0, 0x01, 0x2D, 0xE9, 0xF0, 0x00, 0x9C, 0xE8, 0x01, 0xC0,
109            0x43, 0xE0, 0x07, 0xC0, 0x0C, 0xE2, 0x00, 0x00, 0x5C, 0xE3,
110        ];
111        const B_COMB: [u8; 24] = [
112            0x30, 0x00, 0x2D, 0xE9, 0x01, 0x40, 0xA0, 0xE3, 0x14, 0x43, 0xA0, 0xE1, 0x01, 0x40,
113            0x44, 0xE2, 0xF0, 0x50, 0xD0, 0xE1, 0xF0, 0xC0, 0xD1, 0xE1,
114        ];
115
116        if find_block(rom, &A_CONV, &A_MIX, 0xC0, &A_COMB, 0x1E0) {
117            return Some(Self::Sma23Shift8);
118        }
119        if find_block(rom, &B_CONV, &B_MIX, 0x120, &B_COMB, 0x218) {
120            return Some(Self::Sma4Mla);
121        }
122        None
123    }
124}

126/// Returns `true` when `conv`, `mix` (at `mix_delta`), and `comb` (at
127/// `comb_delta`) signatures all occur at consistent relative offsets.
128fn find_block(rom: &[u8], conv: &[u8; 24], mix: &[u8; 24], mix_delta: usize, comb: &[u8; 24], comb_delta: usize) -> bool {
129    let mut start = 0usize;
130    while let Some(rel) = find_subslice(&rom[start..], conv) {
131        let base = start + rel;
132        if base + comb_delta + comb.len() <= rom.len()
133            && rom[base + mix_delta..base + mix_delta + mix.len()] == mix[..]
134            && rom[base + comb_delta..base + comb_delta + comb.len()] == comb[..]
135        {
136            return true;
137        }
138        start = base + 4;
139    }
140    false
141}

143/// Find the first occurrence of `needle` in `haystack`.
144#[inline]
145fn find_subslice(haystack: &[u8], needle: &[u8]) -> Option<usize> {
146    if needle.is_empty() || needle.len() > haystack.len() {
147        return None;
148    }
149    haystack.windows(needle.len()).position(|w| w == needle)
150}

152#[cfg(test)]
153mod tests {
154    use super::*;
155
156    #[test]
157    fn revision_metadata_is_consistent() {
158        assert!(EadSmaVersion::Sma23Shift8.mixer_shifts_product());
159        assert!(!EadSmaVersion::Sma4Mla.mixer_shifts_product());
160        assert_eq!(EadSmaVersion::Sma23Shift8.corpus_rom_count(), 12);
161        assert_eq!(EadSmaVersion::Sma4Mla.corpus_rom_count(), 23);
162        assert_eq!(EadSmaVersion::detect_from_bytes(&[]), None);
163        assert_eq!(EadSmaVersion::detect_from_bytes(&[0u8; 4096]), None);
164    }
