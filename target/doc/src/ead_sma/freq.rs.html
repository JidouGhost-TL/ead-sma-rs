1//! Fixed-point pitch, phase, panning, and division helpers for Nintendo EAD/SRD.
2
3/// GBA ARM7TDMI master clock frequency in Hz (`2^24` = 16,777,216 Hz).
4pub const GBA_CLOCK_HZ: u32 = 16_777_216;
5
6/// Number of samples rendered per DirectSound DMA frame (`0xB0` = 176).
7///
8/// The retail frame renderer always mixes exactly this many frames into each
9/// planar `i16` channel buffer (`0x160` halfwords per channel) before the
10/// saturating converter packs the result for the DMA FIFO.
11pub const DMA_FRAME_SAMPLES: usize = 0xB0;
12
13/// Unit step (`1.0x` playback speed) in the driver's Q8 phase format (`1 << 8`).
14///
15/// The ARM voice mixer indexes sample data with `phase >> 8`, so the low 8
16/// bits form the fractional phase and are discarded (nearest-neighbour —
17/// neither revision interpolates).
18pub const Q8_ONE: u32 = 1 << 8;
19
20/// Compute the Q8 phase increment for playing a sample recorded at
21/// `sample_rate_hz` through a mixer running at `mix_rate_hz`.
22#[inline]
23#[must_use]
24pub const fn freq_to_step_q8(sample_rate_hz: u32, mix_rate_hz: u32) -> u32 {
25    if mix_rate_hz == 0 {
26        return 0;
27    }
28    (((sample_rate_hz as u64) << 8) / (mix_rate_hz as u64)) as u32
29}
30
31/// Convert a Q8 phase increment back to an effective playback rate in Hz.
32#[inline]
33#[must_use]
34pub fn step_q8_to_freq(step_q8: u32, mix_rate_hz: u32) -> f64 {
35    (step_q8 as f64) * (mix_rate_hz as f64) / (Q8_ONE as f64)
36}
37
38/// Compute the exact left/right mixer gains `(vol_l, vol_r)` used by the
39/// retail Thumb voice wrapper.
40///
41/// The wrapper computes, with 32-bit wrapping multiplication followed by a
42/// `<< 16 >> 23` logical-shift pair (equivalent to masking to 16 bits and
43/// shifting right by 7):
44/// ```text
45/// vol_l = (((127 - pan) * volume) & 0xFFFF) >> 7
46/// vol_r = ((pan * volume) & 0xFFFF) >> 7
47/// ```
48/// where `volume` is the 16-bit envelope/master volume and `pan` is the
49/// 8-bit pan byte (`0` = hard left, `127` = hard right in normal use).
50#[inline]
51#[must_use]
52pub const fn pan_gains(volume: u16, pan: u8) -> (u32, u32) {
53    let vol = volume as u32;
54    let prod_l = ((127i32 - (pan as i32)) as u32).wrapping_mul(vol);
55    let prod_r = (pan as u32).wrapping_mul(vol);
56    ((prod_l & 0xFFFF) >> 7, (prod_r & 0xFFFF) >> 7)
57}
58
59/// Unsigned 32-bit division exactly as implemented by the driver's ARM
60/// helper routine: returns `numerator / denominator`, or `0` when
61/// `denominator` is zero.
62#[inline]
63#[must_use]
64pub const fn arm_udiv(numerator: u32, denominator: u32) -> u32 {
65    match numerator.checked_div(denominator) {
66        Some(v) => v,
67        None => 0,
68    }
69}
70
71/// Signed 32-bit division exactly as implemented by the driver's ARM helper
72/// routine: returns `numerator / denominator` (truncating toward zero), or
73/// `0` when `denominator` is zero. The `i32::MIN / -1` overflow case wraps
74/// to `i32::MIN`, matching the helper's shift/subtract sequence.
75#[inline]
76#[must_use]
77pub const fn arm_sdiv(numerator: i32, denominator: i32) -> i32 {
78    match numerator.checked_div(denominator) {
79        Some(v) => v,
80        // Division by zero (ARM helper returns 0) and the i32::MIN / -1
81        // overflow (the helper's shift/subtract sequence wraps to i32::MIN).
82        None => {
83            if denominator == 0 {
84                0
85            } else {
86                i32::MIN
87            }
88        }
89    }
90}

92/// Compute the exact DirectSound hardware sample rate in Hz for a GBA
93/// Timer0 reload value.
94#[inline]
95#[must_use]
96pub const fn rate_from_timer_reload(reload: u16) -> u32 {
97    let ticks = 65_536u32 - (reload as u32);
98    (GBA_CLOCK_HZ + (ticks / 2)) / ticks
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    #[test]
106    fn pitch_pan_and_division_helpers_work() {
107        assert_eq!(freq_to_step_q8(10_512, 10_512), Q8_ONE);
108        assert_eq!(freq_to_step_q8(21_024, 10_512), Q8_ONE * 2);
109        assert_eq!(freq_to_step_q8(5_256, 10_512), Q8_ONE / 2);
110        assert_eq!(step_q8_to_freq(Q8_ONE, 10_512), 10_512.0);
111        assert_eq!(pan_gains(128, 0), (127, 0));
112        assert_eq!(pan_gains(128, 127), (0, 127));
113        assert_eq!(pan_gains(256, 64), (126, 128));
114        assert_eq!(arm_udiv(100, 7), 14);
115        assert_eq!(arm_udiv(100, 0), 0);
116        assert_eq!(arm_sdiv(-100, 7), -14);
117        assert_eq!(arm_sdiv(100, 0), 0);
118        assert_eq!(arm_sdiv(i32::MIN, -1), i32::MIN);
119        assert_eq!(DMA_FRAME_SAMPLES, 176);
120    }
121}
